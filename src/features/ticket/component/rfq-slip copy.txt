


'use client';
import CustomButton from '@/components/atoms/custom-button';
import { Heading } from '@/components/atoms/heading';
import { Text } from '@/components/atoms/text';
import {
  BuyerDto,
  TransporterDto,
} from '@/features/authentication/types/onboarding.types';
import { TruckDto } from '@/features/transporter-dashboard/types/truck.type';
import { TruckOrderDto } from '@/types/truck-order.types';
import { DepotHubDto } from '@/types/depot-hub.types';
import { ProductDto } from '@/types/product.types';
import { formatDateDashTime } from '@/utils/formatDate';
import { formatNumber } from '@/utils/formatNumber';
import { FGTruckFill } from '@fg-icons';
import { Download } from 'lucide-react';
import { Sora } from 'next/font/google';
import React, { useContext, useState } from 'react';
import html2canvas from 'html2canvas';
import jsPDF from 'jspdf';
import { UserType } from '@/types/user.types';
import { AuthContext } from '@/contexts/AuthContext';
import { useServiceFees } from '@/hooks/useServiceFees.hook';

const sora = Sora({ subsets: ['latin'] });

// Unique, authentic, currency-like RFQ slip design
const RfqSlipCurrency: React.FC<{ truckOrder: TruckOrderDto }> = ({
  truckOrder,
}) => {
  const ticketRef = React.useRef<HTMLDivElement>(null);
  const [isDownloading, setIsDownloading] = useState<boolean>(false);
  const { user } = useContext(AuthContext);
  const userRole = user?.data?.role;
  const { serviceFees, isLoading: isLoadingFees } = useServiceFees();

  // Calculate platform service charges
  const transportCost = truckOrder.price;
  const transporterServiceCharge =
    transportCost * serviceFees.transporterServiceFee;
  const buyerServiceCharge = transportCost * serviceFees.traderServiceFee;

  // Download logic (same as before, but can be customized for new design)
  const handleDownload = async () => {
    setIsDownloading(true);
    if (!ticketRef.current) {
      setIsDownloading(false);
      return;
    }
    try {
      const container = document.createElement('div');
      container.style.position = 'absolute';
      container.style.left = '-9999px';
      container.style.top = '0';
      document.body.appendChild(container);
      const ticketClone = ticketRef.current.cloneNode(true) as HTMLElement;
      container.appendChild(ticketClone);
      await new Promise((resolve) => setTimeout(resolve, 1000));
      const canvas = await html2canvas(ticketClone, {
        scale: 2,
        useCORS: true,
        backgroundColor: null,
        logging: false,
      });
      document.body.removeChild(container);
      const canvasWidth = canvas.width;
      const canvasHeight = canvas.height;
      const pdfWidth = canvasWidth * 0.264583;
      const pdfHeight = canvasHeight * 0.264583;
      const pdf = new jsPDF({
        orientation: pdfWidth > pdfHeight ? 'l' : 'p',
        unit: 'mm',
        format: [pdfWidth, pdfHeight],
      });
      const imgData = canvas.toDataURL('image/png', 1.0);
      pdf.addImage(imgData, 'PNG', 0, 0, pdfWidth, pdfHeight);
      pdf.save(`rfq-currency-${truckOrder.trackingId}.pdf`);
    } catch (error) {
      console.error('Error generating PDF:', error);
    } finally {
      setIsDownloading(false);
    }
  };
  // Currency-like design: gold foil, watermark, serial number, hologram effect, microtext, security border
  return (
    <div
      ref={ticketRef}
      className="relative w-[480px] mx-auto bg-gradient-to-br from-[#F5E9C8] via-[#E6D7B0] to-[#C9B37F] border-8 border-double border-[#D4AF37] rounded-[32px] shadow-2xl overflow-hidden p-0 font-sans"
      style={{
        backgroundImage:
          "url('/images/currency-watermark.svg'), url('/images/Subtract.svg')",
        backgroundRepeat: 'no-repeat, no-repeat',
        backgroundPosition: 'center center, left bottom',
        backgroundSize: '320px, cover',
        // Watermark design tip: create a semi-transparent SVG with your logo, gold color, and some opacity (e.g. 0.08-0.15). Place it at /public/images/currency-watermark.svg
      }}
    >
      {/* Security border and hologram */}
      <div className="absolute top-0 left-0 w-full h-full pointer-events-none z-0">
        <div className="w-full h-full border-4 border-dashed border-[#BFAF7A] rounded-[28px] opacity-40" />
        <div className="absolute right-6 top-6 w-16 h-16 bg-gradient-to-br from-[#D4AF37] to-[#fffbe6] rounded-full shadow-lg opacity-70 animate-pulse" />
      </div>
      {/* Serial number and logo */}
      <div className="flex justify-between items-center px-8 pt-8 pb-2 relative z-10">
        <img
          src="/images/logo_gold.png"
          alt="Fuelsgate Logo"
          className="h-12 w-auto drop-shadow-lg"
        />
        <Text
          variant="pm"
          color="text-[#BFAF7A]"
          fontWeight="bold"
          classNames="tracking-widest text-xs"
        >
          Serial No: {truckOrder.trackingId}
        </Text>
      </div>
      {/* Main details */}
      <div className="px-8 py-4 relative z-10">
        <Heading
          variant="h3"
          color="text-gold"
          fontWeight="bold"
          classNames="mb-2 text-center tracking-wider"
        >
          FUELS GATE CURRENCY TICKET
        </Heading>
        <div className="grid grid-cols-2 gap-x-8 gap-y-3 text-sm mb-4">
          <div>
            <Text variant="pxs" color="text-gold" classNames="mb-1">
              Buyer
            </Text>
            <Text variant="pxs" color="text-deep-gray-100" fontWeight="bold">
              {
                ((truckOrder.buyerId as BuyerDto)?.userId as UserType)
                  ?.firstName
              }{' '}
              {((truckOrder.buyerId as BuyerDto)?.userId as UserType)?.lastName}
            </Text>
          </div>
          <div>
            <Text variant="pxs" color="text-gold" classNames="mb-1">
              Transporter
            </Text>
            <Text variant="pxs" color="text-deep-gray-100" fontWeight="bold">
              {(truckOrder.profileId as TransporterDto)?.companyName}
            </Text>
          </div>
          <div>
            <Text variant="pxs" color="text-gold" classNames="mb-1">
              Product Type
            </Text>
            <Text
              classNames="uppercase"
              variant="pxs"
              color={'text-deep-gray-100'}
              fontWeight="bold"
            >
              {
                ((truckOrder.truckId as TruckDto)?.productId as ProductDto)
                  ?.value
              }
            </Text>
          </div>
          <div>
            <Text variant="pxs" color="text-gold" classNames="mb-1">
              Truck Number
            </Text>
            <Text variant="pxs" color="text-deep-gray-100" fontWeight="bold">
              {(truckOrder.truckId as TruckDto)?.truckNumber}
            </Text>
          </div>
          <div>
            <Text variant="pxs" color="text-gold" classNames="mb-1">
              Hub
            </Text>
            <Text variant="pxs" color="text-deep-gray-100" fontWeight="bold">
              {
                ((truckOrder.truckId as TruckDto)?.depotHubId as DepotHubDto)
                  ?.name
              }
            </Text>
          </div>
          <div>
            <Text variant="pxs" color="text-gold" classNames="mb-1">
              Loading Depot
            </Text>
            <Text variant="pxs" color="text-deep-gray-100" fontWeight="bold">
              {truckOrder.loadingDepot}
            </Text>
          </div>
          <div className="col-span-2">
            <Text variant="pxs" color="text-gold" classNames="mb-1">
              Destination
            </Text>
            <Text variant="pxs" color="text-deep-gray-100" fontWeight="bold">
              {truckOrder.destination}, {truckOrder.city}, {truckOrder.state}
            </Text>
          </div>
          <div>
            <Text variant="pxs" color="text-gold" classNames="mb-1">
              Truck Capacity
            </Text>
            <Text variant="pxs" color="text-deep-gray-100" fontWeight="bold">
              {formatNumber((truckOrder.truckId as TruckDto)?.capacity)} Ltrs
            </Text>
          </div>
          <div>
            <Text variant="pxs" color="text-gold" classNames="mb-1">
              Loading Date
            </Text>
            <Text variant="pxs" color="text-deep-gray-100" fontWeight="bold">
              {truckOrder.loadingDate
                ? formatDateDashTime(truckOrder.loadingDate.toString())
                : 'TBD'}
            </Text>
          </div>
          <div className="flex flex-col">
            <Text variant="pxs" color="text-gold" classNames="mb-1">
              Status
            </Text>
            <div className="flex items-center gap-2">
              {/* Truck icon with product color */}
              <span className="inline-flex items-center justify-center mr-1">
                <FGTruckFill
                  color={
                    typeof truckOrder.truckId === 'object'
                      ? (truckOrder.truckId as TruckDto)?.productId?.color
                      : undefined
                  }
                  height={18}
                  width={18}
                />
              </span>
              <Text variant="pxs" color="text-deep-gray-100" fontWeight="bold">
                {truckOrder.status}
              </Text>
            </div>
          </div>
          <div className="flex flex-col">
            {/* Completed time in second column, only if completed */}
            {truckOrder.status?.toLowerCase() === 'completed' &&
              truckOrder.updatedAt && (
                <>
                  <Text variant="pxs" color="text-gold" classNames="mb-1">
                    Completed Time
                  </Text>
                  <Text
                    variant="pxs"
                    color="text-deep-gray-100"
                    fontWeight="bold"
                  >
                    {formatDateDashTime(truckOrder.updatedAt.toString())}
                  </Text>
                </>
              )}
          </div>
        </div>
        {/* Microtext security line */}
        <div className="w-full text-center text-[8px] text-[#BFAF7A] tracking-tight mb-2 opacity-80 select-none">
          FUELS GATE RESOURCES LTD • SECURE CURRENCY TICKET • AUTHENTICITY
          GUARANTEED • {truckOrder.trackingId}
        </div>
        {/* Watermark and hologram effect (visual only, no text label) */}
        <div className="flex justify-between items-center mb-4">
          {/* Hologram shimmer: gold animated circle for authenticity */}
          <div className="flex items-center gap-2">
            <span className="w-6 h-6 rounded-full bg-gradient-to-br from-[#D4AF37] to-[#fffbe6] shadow-md opacity-80 animate-pulse border-2 border-[#BFAF7A]" />
          </div>
          {/* Watermark: faint logo background, see backgroundImage above. To use a watermark image, create a semi-transparent SVG or PNG and place it in /public/images/currency-watermark.svg */}
        </div>
        {/* Amounts and payment info */}
        <div className="bg-gradient-to-r from-[#F5E9C8]/60 to-[#C9B37F]/40 rounded-xl border-2 border-[#BFAF7A] p-4 mb-2">
          <div className="flex flex-wrap items-center justify-between">
            <Text variant="pm" fontWeight="bold" color="text-[#BFAF7A]">
              Transport Fee
            </Text>
            <Heading
              variant="h4"
              color="text-[#BFAF7A]"
              fontFamily={sora.className}
              fontWeight="bold"
            >
              ₦{formatNumber(transportCost, true)}
            </Heading>
          </div>
        </div>
        <div className="bg-gradient-to-r from-[#F5E9C8]/40 to-[#C9B37F]/20 rounded-xl border border-[#BFAF7A] p-3 mb-2 opacity-80">
          <div className="flex flex-wrap items-center justify-between">
            <Text variant="pm" fontWeight="medium" color="text-[#BFAF7A]">
              Platform Service Charge
            </Text>
            <Heading
              variant="h5"
              color="text-[#BFAF7A]"
              fontFamily={sora.className}
              fontWeight="medium"
            >
              {userRole === 'transporter'
                ? isLoadingFees
                  ? 'Loading...'
                  : `₦${formatNumber(transporterServiceCharge, true)}`
                : isLoadingFees
                ? 'Loading...'
                : `₦${formatNumber(buyerServiceCharge, true)}`}
            </Heading>
          </div>
        </div>
        {/* Bank details */}
        <div className="grid grid-cols-2 gap-x-8 gap-y-3 mt-2">
          {userRole === 'transporter' ? (
            <>
              <div>
                <Text
                  variant="pm"
                  color="text-gold"
                  classNames="mb-1 font-bold text-base"
                >
                  Bank Name
                </Text>
                <Text
                  variant="pm"
                  color="text-deep-gray-100"
                  fontWeight="bold"
                  classNames="text-lg"
                >
                  FIDELITY BANK PLC
                </Text>
              </div>
              <div>
                <Text
                  variant="pm"
                  color="text-gold"
                  classNames="mb-1 font-bold text-base"
                >
                  Account Name
                </Text>
                <Text
                  variant="pm"
                  color="text-deep-gray-100"
                  fontWeight="bold"
                  classNames="text-lg"
                >
                  FUELS GATE RESOURCES LTD
                </Text>
              </div>
              <div className="col-span-2">
                <Text
                  variant="pm"
                  color="text-gold"
                  classNames="mb-1 font-bold text-base"
                >
                  Account Number
                </Text>
                <Text
                  variant="pm"
                  color="text-deep-gray-100"
                  fontWeight="bold"
                  classNames="text-lg"
                >
                  5540001642
                </Text>
              </div>
            </>
          ) : (
            <>
              <div>
                <Text
                  variant="pm"
                  color="text-gold"
                  classNames="mb-1 font-bold text-base"
                >
                  Bank Name
                </Text>
                <Text
                  variant="pm"
                  color="text-deep-gray-100"
                  fontWeight="bold"
                  classNames="text-lg"
                >
                  ZENITH BANK PLC
                </Text>
              </div>
              <div>
                <Text
                  variant="pm"
                  color="text-gold"
                  classNames="mb-1 font-bold text-base"
                >
                  Account Name
                </Text>
                <Text
                  variant="pm"
                  color="text-deep-gray-100"
                  fontWeight="bold"
                  classNames="text-lg"
                >
                  FUELS GATE RESOURCES LTD
                </Text>
              </div>
              <div className="col-span-2">
                <Text
                  variant="pm"
                  color="text-gold"
                  classNames="mb-1 font-bold text-base"
                >
                  Account Number
                </Text>
                <Text
                  variant="pm"
                  color="text-deep-gray-100"
                  fontWeight="bold"
                  classNames="text-lg"
                >
                  1013478130
                </Text>
              </div>
            </>
          )}
        </div>
      </div>
      {/* Footer authenticity */}
      <div className="w-full text-center text-[11px] text-deep-gray-100 tracking-widest py-2 bg-[#F5E9C8]/80 border-t border-[#BFAF7A] select-none font-bold">
        This ticket is an official Fuels Gate Resources Ltd. currency.
        Counterfeiting is a crime. All rights reserved.
      </div>
      <CustomButton
        variant="primary"
        bgColor="bg-gold hover:bg-[#C9B37F]"
        color="text-deep-gray-300"
        height="h-12"
        fontWeight="bold"
        label="Download Ticket"
        loading={isDownloading}
        leftIcon={<Download />}
        onClick={handleDownload}
      />
    </div>
  );
};

export default RfqSlipCurrency;
